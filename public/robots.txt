# Zugelassen ist die Google-Suche. Alles andere -- SEO-Werkzeuge, Scraper,
# Trainingscrawler, Archivdienste -- ist ausgeschlossen.
#
# Wichtig zu wissen: robots.txt ist eine Bitte, keine Sperre. Wer sich nicht
# daran hält, wird davon nicht aufgehalten. Serverseitig abgewiesen wird seit
# der Deinstallation der Netlify-Integration "User Agent Blocker" niemand mehr;
# diese Datei ist damit die einzige Stelle, an der die Liste steht.
#
# Bing, DuckDuckGo und alle übrigen Suchmaschinen fallen unter die
# Schlussregel und sind damit ausgeschlossen. Das war eine Entscheidung, keine
# Nachlässigkeit -- wer dort gefunden werden will, trägt Bingbot oben ein.
# ---------------------------------------------------------------------------

# --- Google-Suche ----------------------------------------------------------
User-agent: Googlebot
User-agent: Googlebot-Image
User-agent: Googlebot-News
User-agent: Google-InspectionTool
User-agent: Storebot-Google
Allow: /

# Google-Extended steuert ausschließlich das Training von Gemini, nicht die
# Suche. Ausschließen kostet also keine Sichtbarkeit.
User-agent: Google-Extended
Disallow: /

# --- ChatGPT ---------------------------------------------------------------
# Serverseitig derzeit nicht freigegeben, deshalb hier ebenfalls nicht. Soll
# ChatGPT die Seite in seinen Antworten zitieren dürfen, sind es zwei
# Schritte: OAI-SearchBot und ChatGPT-User in der Netlify-Integration
# freigeben, dann hier die vier Zeilen unter diesem Absatz einkommentieren.
# Die Bremse gehört dazu -- eine Anfrage alle zehn Sekunden reicht für eine
# Website dieser Größe.
#
# User-agent: OAI-SearchBot
# Allow: /
# Crawl-delay: 10
#
# User-agent: ChatGPT-User
# Allow: /
# Crawl-delay: 10

# GPTBot sammelt Trainingsmaterial. Das ist etwas anderes als eine Antwort auf
# eine Frage und bleibt in jedem Fall ausgeschlossen.
User-agent: GPTBot
Disallow: /

# --- Vorschaubilder in Chats und Netzwerken ---------------------------------
# Diese Dienste crawlen nicht, sie holen genau die eine Seite, deren Adresse
# jemand gerade teilt, und lesen daraus Titel und Bild. Ohne Freigabe bleibt
# ein geteilter Link eine nackte Textzeile -- in WhatsApp, LinkedIn, Slack und
# überall sonst. Das kostet Sichtbarkeit, ohne irgendetwas zu schützen.
User-agent: LinkedInBot
User-agent: Twitterbot
User-agent: facebookexternalhit
User-agent: WhatsApp
User-agent: Slackbot
User-agent: Slackbot-LinkExpanding
User-agent: Discordbot
User-agent: TelegramBot
User-agent: SkypeUriPreview
Allow: /
